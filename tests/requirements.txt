# Pinned deliberately (proposals/17). These were unpinned, which meant a
# Playwright or pytest major release could break CI with no change from
# this repo -- and the failure would look like a regression in the app.
#
# To bump: change a version here, run the suite locally, and land the
# bump as its own commit so a behavioural change and a toolchain change
# are never in the same diff.
pytest==9.1.1
playwright==1.56.0

# Parallel test execution (-n auto). The `browser` and `base_url`
# fixtures are session-scoped, so each worker gets its own browser and
# its own static-file server, which is correct rather than merely
# tolerated: tests share nothing but the files on disk.
pytest-xdist==3.8.0

# axe-core for tests/test_accessibility.py (proposals/18). The wheel
# bundles axe.min.js, so the accessibility job needs no CDN fetch and no
# Node -- test.yml's `test` job is deliberately both offline and
# Node-free, and a vendored 550 KB axe.min.js in the repo was the only
# other way to keep it that way.
axe-core-python==0.1.0
